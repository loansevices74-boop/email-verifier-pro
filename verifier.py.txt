    def __init__(self, timeout: int = 5):
        self.timeout = timeout
        self.resolver = dns.resolver.Resolver()
        self.resolver.timeout = timeout
        self.resolver.lifetime = timeout
        # Use multiple public DNS servers for reliability
        self.resolver.nameservers = ['8.8.8.8', '1.1.1.1', '9.9.9.9']

    def _check_syntax(self, email: str) -> Tuple[bool, str]:
        """RFC 5322 syntax validation."""
        if not email or '@' not in email:
            return False, "Missing @ symbol"

        local, _, domain = email.partition('@')

        if not local or not domain:
            return False, "Empty local or domain part"

        if len(email) > 254:
            return False, "Email too long (>254 chars)"

        if len(local) > 64:
            return False, "Local part too long (>64 chars)"

        if len(domain) > 253:
            return False, "Domain too long (>253 chars)"

        # RFC 5322 pattern
        pattern = r'^[a-zA-Z0-9.!#$%&\'*+/=?^_`{|}~-]+@[a-zA-Z0-9](?:[a-zA-Z0-9-]{0,61}[a-zA-Z0-9])?(?:\.[a-zA-Z0-9](?:[a-zA-Z0-9-]{0,61}[a-zA-Z0-9])?)*$'

        if not re.match(pattern, email):
            return False, "Invalid characters"

        # Check for consecutive dots
        if '..' in local or '..' in domain:
            return False, "Consecutive dots"

        if local.startswith('.') or local.endswith('.'):
            return False, "Local part starts/ends with dot"

        return True, "Valid syntax"

    def _check_dns(self, domain: str) -> Tuple[bool, List[str], str]:
        """Check MX and A records."""
        try:
            # Try MX first
            try:
                mx_records = self.resolver.resolve(domain, 'MX')
                mx_hosts = sorted(
                    [(r.preference, str(r.exchange).rstrip('.'))
                     for r in mx_records]
                )
                hosts = [h for _, h in mx_hosts]
                return True, hosts, "MX records found"
            except dns.resolver.NoAnswer:
                pass
            except dns.resolver.NXDOMAIN:
                return False, [], "Domain does not exist"

            # Fallback to A record
            try:
                a_records = self.resolver.resolve(domain, 'A')
                if a_records:
                    return True, [], "A record found (no MX)"
            except:
                pass

            return False, [], "No MX or A records"

        except dns.resolver.Timeout:
            return False, [], "DNS timeout"
        except dns.exception.DNSException as e:
            return False, [], f"DNS error: {str(e)}"
        except Exception as e:
            return False, [], f"Error: {str(e)}"

    def _detect_provider(self, mx_hosts: List[str], domain: str) -> str:
        """Identify email provider from MX records."""
        combined = ' '.join(mx_hosts).lower() + ' ' + domain.lower()

        for provider, patterns in self.PROVIDER_PATTERNS.items():
            if any(p in combined for p in patterns):
                return provider

        return 'unknown'

    def _check_typo(self, domain: str) -> str:
        """Suggest corrected domain if typo detected."""
        domain_lower = domain.lower()
        return self.TYPO_DOMAINS.get(domain_lower, '')

    def _is_disposable(self, domain: str) -> bool:
        """Check if domain is disposable."""
        return domain.lower() in self.DISPOSABLE_DOMAINS

    def _is_role_account(self, local_part: str) -> bool:
        """Check if it's a role account."""
        return local_part.lower() in self.ROLE_ACCOUNTS

    def verify(self, email: str) -> Dict:
        """
        Main verification method.
        Returns comprehensive result dict.
        """
        email = email.strip().lower()
        start_time = time.time()

        result = {
            'email': email,
            'status': 'UNKNOWN',
            'confidence': 0,
            'reason': '',
            'domain': '',
            'mx_records': '',
            'provider': 'unknown',
            'is_disposable': False,
            'is_role': False,
            'is_catch_all': None,
            'typo_suggestion': '',
            'syntax_valid': False,
            'dns_valid': False,
            'timestamp': datetime.now().isoformat(),
            'processing_time_ms': 0
        }

        # Layer 1: Syntax
        syntax_ok, syntax_msg = self._check_syntax(email)
        result['syntax_valid'] = syntax_ok

        if not syntax_ok:
            result.update({
                'status': 'INVALID',
                'confidence': 0,
                'reason': f'Syntax error: {syntax_msg}'
            })
            return result

        # Extract parts
        local_part, _, domain = email.partition('@')
        result['domain'] = domain

        # Layer 2: Typo detection
        typo = self._check_typo(domain)
        result['typo_suggestion'] = f"{local_part}@{typo}" if typo else ''

        # Layer 3: DNS validation
        dns_ok, mx_hosts, dns_msg = self._check_dns(domain)
        result['dns_valid'] = dns_ok
        result['mx_records'] = '; '.join(mx_hosts[:5]) if mx_hosts else 'none'

        if not dns_ok:
            result.update({
                'status': 'INVALID',
                'confidence': 5,
                'reason': f'DNS failed: {dns_msg}'
            })
            return result

        # Layer 4: Provider detection
        provider = self._detect_provider(mx_hosts, domain)
        result['provider'] = provider

        # Layer 5: Content flags
        is_disposable = self._is_disposable(domain)
        is_role = self._is_role_account(local_part)
        result['is_disposable'] = is_disposable
        result['is_role'] = is_role

        # Layer 6: Final decision
        if is_disposable:
            result.update({
                'status': 'RISKY',
                'confidence': 20,
                'reason': 'Disposable email domain'
            })
        elif is_role:
            result.update({
                'status': 'RISKY',
                'confidence': 35,
                'reason': 'Role account (not personal mailbox)'
            })
        elif provider in ['gmail', 'outlook', 'yahoo', 'hotmail', 'icloud', 'protonmail']:
            # Major providers accept-all at SMTP level
            result.update({
                'status': 'LIKELY_VALID',
                'confidence': 75,
                'reason': f'{provider.capitalize()} domain - mailbox existence cannot be verified without SMTP'
            })
        elif typo:
            result.update({
                'status': 'RISKY',
                'confidence': 40,
                'reason': f'Possible typo - did you mean {typo}?'
            })
        else:
            # Business/custom domain with valid MX
            result.update({
                'status': 'VALID',
                'confidence': 85,
                'reason': 'Domain accepts mail, no red flags'
            })

        result['processing_time_ms'] = int((time.time() - start_time) * 1000)
        return result

    def verify_batch(self, emails: List[str], progress_callback=None) -> List[Dict]:
        """Verify multiple emails with optional progress callback."""
        results = []
        total = len(emails)

        for i, email in enumerate(emails):
            result = self.verify(email)
            results.append(result)

            if progress_callback:
                progress_callback(i + 1, total, result)

        return results
